const http = require('http'); const fs = require('fs'); const path = require('path'); const crypto = require('crypto'); const WebSocket = require('ws'); const PORT = process.env.PORT || 3000; const DB_FILE = path.join(__dirname, 'flappy-dusk-db.json'); const GAME_FILE = path.join(__dirname, 'index.html'); let db = { users: {} }; try { db = JSON.parse(fs.readFileSync(DB_FILE, 'utf8')); } catch {} if (!db.users) db.users = {}; const sessions = new Map(); const rooms = new Map(); function saveDb() { fs.writeFileSync(DB_FILE, JSON.stringify(db, null, 2)); } function hashPassword(password, salt) { return crypto.scryptSync(password, salt, 64).toString('hex'); } function makePassword(password) { const salt = crypto.randomBytes(16).toString('hex'); return { salt, hash: hashPassword(password, salt) }; } function verifyPassword(password, rec) { return crypto.timingSafeEqual( Buffer.from(hashPassword(password, rec.salt), 'hex'), Buffer.from(rec.hash, 'hex') ); } function token() { return crypto.randomBytes(32).toString('hex'); } function auth(req) { const h = req.headers.authorization || ''; if (!h.startsWith('Bearer ')) return null; const name = sessions.get(h.slice(7)); return name ? db.users[name] : null; } function body(req) { return new Promise((resolve, reject) => { let s = ''; req.on('data', c => { s += c; }); req.on('end', () => { try { resolve(JSON.parse(s || '{}')); } catch (e) { reject(e); } }); }); } function send(res, status, data) { res.writeHead(status, { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }); res.end(JSON.stringify(data)); } const defaultProgress = () => ({ coins: 0, totalPipesPassed: 0, ownedBirds: { classic: true }, selectedBird: 'classic', best: 0, updatedAt: Date.now() }); const server = http.createServer(async (req, res) => { if (req.method === 'OPTIONS') { res.writeHead(204, { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'Content-Type,Authorization' }); return res.end(); } try { // CREATE ACCOUNT if (req.url === '/api/signup' && req.method === 'POST') { const b = await body(req); const name = String(b.name || '') .trim() .slice(0, 20); const password = String(b.password || ''); if ( !/^[A-Za-z0-9 _-]{2,20}$/.test(name) || password.length < 4 ) { return send(res, 400, { error: 'Use a 2–20 character name and a 4+ character password.' }); } if (db.users[name]) { return send(res, 409, { error: 'That name is already taken.' }); } const p = makePassword(password); db.users[name] = { password: p, progress: defaultProgress(), createdAt: Date.now() }; saveDb(); const t = token(); sessions.set(t, name); return send(res, 200, { token: t, user: { name }, progress: db.users[name].progress }); } // LOGIN if (req.url === '/api/login' && req.method === 'POST') { const b = await body(req); const name = String(b.name || '').trim(); const password = String(b.password || ''); const u = db.users[name]; if ( !u || !u.password || !verifyPassword(password, u.password) ) { return send(res, 401, { error: 'Name or password is incorrect.' }); } const t = token(); sessions.set(t, name); return send(res, 200, { token: t, user: { name }, progress: u.progress || defaultProgress() }); } // CURRENT ACCOUNT if (req.url === '/api/me' && req.method === 'GET') { const u = auth(req); if (!u) { return send(res, 401, { error: 'Not logged in' }); } return send(res, 200, { user: { name: Object.keys(db.users).find( k => db.users[k] === u ) }, progress: u.progress }); } // SAVE PROGRESS if (req.url === '/api/progress' && req.method === 'PUT') { const u = auth(req); if (!u) { return send(res, 401, { error: 'Not logged in' }); } const b = await body(req); u.progress = { ...defaultProgress(), ...b, ownedBirds: b.ownedBirds || { classic: true }, updatedAt: Date.now() }; saveDb(); return send(res, 200, { ok: true }); } // LEADERBOARD if (req.url === '/api/leaderboard' && req.method === 'GET') { const rows = Object.entries(db.users) .map(([name, u]) => ({ name, pipes: Number(u.progress?.totalPipesPassed) || 0, best: Number(u.progress?.best) || 0 })) .sort( (a, b) => b.pipes - a.pipes || b.best - a.best ) .slice(0, 20); return send(res, 200, { rows }); } // UNKNOWN API if (req.url.startsWith('/api/')) { return send(res, 404, { error: 'Not found' }); } // SERVE THE GAME let file = req.url === '/' ? GAME_FILE : path.join( __dirname, decodeURIComponent( req.url.split('?')[0] ) ); if ( !file.startsWith(__dirname) || !fs.existsSync(file) || fs.statSync(file).isDirectory() ) { return send(res, 404, { error: 'Not found' }); } const ext = path.extname(file); const type = ext === '.html' ? 'text/html' : ext === '.js' ? 'text/javascript' : 'application/octet-stream'; res.writeHead(200, { 'Content-Type': type }); fs.createReadStream(file).pipe(res); } catch (e) { console.error(e); send(res, 500, { error: 'Server error' }); } }); // WEBSOCKET MULTIPLAYER const wss = new WebSocket.Server({ server, path: '/ws' }); wss.on('connection', ws => { ws.on('message', raw => { let m; try { m = JSON.parse(raw); } catch { return; } // JOIN ROOM if (m.type === 'join') { const name = sessions.get(m.token); if (!name) { return ws.send( JSON.stringify({ type: 'error', message: 'Log in first.' }) ); } const room = String(m.room || '') .toUpperCase(); if (!/^[A-Z0-9]{6}$/.test(room)) { return ws.send( JSON.stringify({ type: 'error', message: 'Invalid room code.' }) ); } let r = rooms.get(room); if (!r) { r = { host: null, guest: null, hostName: '', guestName: '' }; rooms.set(room, r); } // HOST if (m.role === 'host' && !r.host) { r.host = ws; r.hostName = name; ws.room = room; ws.role = 'host'; ws.send( JSON.stringify({ type: 'roomJoined', role: 'host' }) ); } // GUEST else if (!r.guest) { r.guest = ws; r.guestName = name; ws.room = room; ws.role = 'guest'; ws.send( JSON.stringify({ type: 'roomJoined', role: 'guest', hostName: r.hostName }) ); if ( r.host && r.host.readyState === 1 ) { r.host.send( JSON.stringify({ type: 'peerJoined', guestName: name }) ); } } // ROOM FULL else { ws.send( JSON.stringify({ type: 'error', message: 'Room is full.' }) ); } } // GUEST INPUT → HOST else if ( m.type === 'input' && ws.role === 'guest' ) { const r = rooms.get(ws.room); if ( r?.host?.readyState === 1 ) { r.host.send( JSON.stringify(m) ); } } // HOST SNAPSHOT → GUEST else if ( (m.type === 'snapshot' || m.type === 'matchOver') && ws.role === 'host' ) { const r = rooms.get(ws.room); if ( r?.guest?.readyState === 1 ) { r.guest.send( JSON.stringify(m) ); } } }); // DISCONNECT ws.on('close', () => { if (!ws.room) return; const r = rooms.get(ws.room); if (!r) return; if (r.host === ws) { r.host = null; } if (r.guest === ws) { r.guest = null; } if (!r.host && !r.guest) { rooms.delete(ws.room); } }); }); // RENDER PORT server.listen( PORT, '0.0.0.0', () => { console.log( `Flappy Dusk online server running on port ${PORT}` ); } );
